-- tabela departamento

depcod integer not null auto_increment primary key,

);

-- tabela funcionario
create table funcionario(
funcod integer not null auto_increment primary key,
funnome varchar(60)not null,
funrg varchar(30)not null,
funcpf varchar(30)not null,
fundtnasc varchar(45)not null,
funend varchar(60)not null,
funfone varchar(18)not null,
funsalario double(7,2)not null,
fundepcod integer not null,
foreign key(fundepcod)references departamento(depcod));

-- tabela cliente
create table cliente(
clicod integer not null auto_increment primary key,
clinome varchar(60)not null,
clirg varchar(30)null,
clicpf varchar(30)null,
clisexo varchar(10)not null,
cliend varchar(60)null,
clifone varchar(18)not null
);

-- tabela grupoproduto
create table categoriaProduto(
ctpcod integer not null auto_increment primary key,
ctpdesc varchar(40)not null
);

create table observacao(
obscod integer not null auto_increment primary key,
obsdesc varchar(100) not null
);

create table tamanho(
tamcod integer not null auto_increment primary key,
tamnome varchar(60) not null,
tampreco double(7,2)not null,
);

create table produtoTamanho(
ptcod integer not null auto_increment,
pttamcod integer not null,
ptprocod integer not null,
foreign key(pttamcod)references tamanho(tamcod),
foreign key(ptprocod)references produto(procod),
primary key(ptcod,pttamcod,ptprocod)
);

-- tabela produto
create table produto(
procod integer not null auto_increment primary key,
pronome varchar(60)not null,
propreco double(7,2)not null,
procusto double(7,2)not null,
prounmed varchar(10)not null,
proest integer not null,
proestm integer not null,
prostatus varchar(40) not null,
proctpcod integer not null,
foreign key(proctpcod)references categoriaProduto(ctpcod)
);

create table categoriaItem(
cticod integer not null auto_increment primary key,
ctinome varchar(40) not null
);

create table item(
itcod integer not null auto_increment primary key,
itnome varchar(60)not null,
itcusto double(7,2) not null,
itunmed varchar(4) not null,
itest double(7,2) not null,
itestm double(7,2) not null,
itvalidade integer not null,
itstatus varchar(40) not null,
itforcod integer not null,
itcticod integer not null,
foreign key(itforcod) references fornecedor(forcod),
foreign key(itcticod) references categoriaItem(cticod)
);

create table itemproduto(
itpcod integer not null auto_increment,
itpprocod integer not null,
itpitcod integer not null,
foreign key(itpprocod) references produto(procod),
foreign key(itpitcod) references item(itcod),
primary key(itpcod,itpprocod,itpitcod)
);

-- tabela usuario
create table usuario(
usucodigo integer not null auto_increment primary key,
usunome varchar(60)not null,
ususenha varchar(40)not null,
usutipo varchar(45)not null
);

create table caixa(
cxcod integer not null auto_increment primary key,
cxdata varchar(60)null,
cxdthabre varchar(60)null,
cxdthfecha varchar(60)null,
cxdesc varchar(60)null,
cxsinicial double(8,2)null,
cxsfinal double(8,2)null,
cxobs varchar(100)null,
cxstatus varchar(40)null
);

create table mesas(
mcod integer not null auto_increment primary key
);

ventotal double(8,2) null,
pagamento varchar(45)null,
vrecebido double(7,2)null,
troco double(7,2)null,
venstatus varchar(40)null,
venmstatus varchar(40)null,

foreign key(vencxcod)references caixa(cxcod),

-- tabela itemvenda
create table itemvenda(

itvprocod integer not null,
itvqtde integer null,

foreign key(itvvencod)references venda(vencod),
foreign key(itvprocod)references produto(procod),
primary key(itvcod,itvvencod,itvprocod)
);
